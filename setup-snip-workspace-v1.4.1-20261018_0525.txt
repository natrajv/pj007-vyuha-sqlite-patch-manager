*--@07. Basic Git commands 101 [01]--*
# Clone an existing remote repository
PS> git clone git@github.com:natrajv/sn025-avn-pcf.git sn025
# Check modified, staged, and untracked files
PS> git status
# Stage all changed files
PS> git add .
# Unstage all staged files
PS> git restore --staged .
# View project's commit history 
PS> git log --oneline
# See exact changes made to unstaged files
PS> git diff
# The standard add → commit → push loop to save progress
# Stage changes for a specific file
PS> git add <filename>
# Stage all modified and new files
PS> git add .
# Save staged snapshot with a message
PS> git commit -m "Brief explanation of changes"
--
# Full sync by running these two commands in sequence:
# Update your local files with code your teammates have pushed
PS> git pull --rebase
# Upload your local commits to the remote server
PS> git push
--Versioning
--Basic steps
PS> git add .
PS> git commit -m "Finish feature for release"
PS> git tag -a v1.0.0 -m "Release version 1.0.0"
PS> git push origin v1.0.0
--
# Git tags used for version management
# Create Tag 
PS> git tag v1.0.0
# Create an Annotated Tag (Recommended)
PS> git tag -a v1.0.0 -m "Release version 1.0.0"
# By default, the git push command does not send tags to remote servers. You must explicitly push them
PS> git push origin v1.0.0
# List all tags in your repository
PS> git tag
# Search for specific versions (e.g., all 1.x versions)
PS> git tag -l "v1.*"
# View details of a specific tag (shows the author, date, and commit information)
PS> git show v1.0.0
# Tag a past commit (if you forgot to tag it earlier, grab the commit hash from git log)
PS> git tag -a v1.0.0 a1b2c3d -m "Release version 1.0.0"
# Interrupt to stop lengthy display of information
PS> q #Press-"q" to Quit and exit back to the terminal prompt.
# Untracking folders, files 
PS> git rm --cached -r .arch .versions .rback
PS> git commit -m "Untracked folders"
PS> git push
